package app.work.server.activity

import org.jetbrains.exposed.v1.jdbc.Database
import kotlinx.datetime.LocalDate
import app.work.domain.Activity
import app.work.domain.ActivityKind
import app.work.shared.ActivityStore

interface Days {
    suspend fun load(date: String): DayResponse
}

class ExposedDaysStore(
    private val database: Database,
) : Days {
    override suspend fun load(date: String): DayResponse {
        val localDate = LocalDate.parse(date)
            .orThrow(IllegalArgumentException("Invalid date format"))
        
        val logs = ActivityStore.find(ActivityKind.Log, localDate)
        val schedules = ActivityStore.find(ActivityKind.Schedule, localDate)

        return DayResponse(logs = logs, schedules = schedules)
    }
}
