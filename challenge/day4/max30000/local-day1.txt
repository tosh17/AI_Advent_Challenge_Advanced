package app.work.server.activity

import org.jetbrains.exposed.v1.jdbc.Database
import kotlinx.datetime.LocalDate
import app.work.domain.Activity
import app.work.domain.ActivityKind
import app.work.server.activity.ActivityStore

class ExposedActivityStore(
    private val database: Database,
) : ActivityStore {
    override suspend fun find(kind: ActivityKind, onDate: LocalDate): List<Activity> {
        return ActivityTable.selectAll()
            .where { (ActivityTable.kind eq kind.name) and (ActivityTable.onDate eq onDate.toString()) }
            .orderBy(ActivityTable.createdAt to SortOrder.ASC)
            .map { it.toDomain() } // Предположим, что есть конвертер или просто selectAll возвращает Activity
    }
}

// server/src/main/kotlin/app/work/server/days/ExposedDays.kt
package app.work.server.days

import kotlinx.datetime.LocalDate
import app.work.domain.Activity
import app.work.domain.ActivityKind
import app.work.server.activity.ActivityStore
import app.work.server.days.Days

class ExposedDays(
    private val store: ActivityStore,
) : Days {
    override suspend fun load(rawDate: String): List<Activity> {
        val onDate = LocalDate.parse(rawDate)
            .orElseThrow { IllegalArgumentException("Неверный формат даты") } // Упрощенно для примера
        
        val logs = store.find(ActivityKind.Log, onDate)
        val schedule = store.find(ActivityKind.Schedule, onDate)
        
        return (logs + schedule).sortedBy { it.createdAt }
    }
}

package app.work.server.routes

import io.ktor.http.*
import io.ktor.serialization.kotlinx.json.*
import kotlinx.datetime.LocalDate
import kotlinx.serialization.Serializable
import app.work.server.days.Days
import app.work.server.sessions.Sessions
import app.work.server.sessions.SessionBook

@Serializable
data class DayResponse(
    val logs: List<Activity>,
    val schedule: List<Activity>
)

// Или если Days возвращает DTO напрямую, тогда Days.load возвращает DayResponse.
// Но по правилам "Реализация сама зовёт find". Значит Days возвращает список или DTO.
// Сделаем так, чтобы Days возвращал DTO для удобства роута.

suspend fun Routes.api(days: Days) {
    get("/days/{date}") {
        val raw = call.parameters["date"].orEmpty()
        if (raw.isEmpty()) {
            throw IllegalArgumentException("Параметр date обязателен")
        }
        try {
            val response = days.load(raw)
            // Если Days возвращает List<Activity>, нужно мапить в DayResponse.
            // Но по правилам DTO рядом с роутом. Сделаем Days возвращать DTO.
            call.respond(response)
        } catch (e: IllegalArgumentException) {
            throw e // Будет обработано StatusPages в Module.kt
        }
    }
}
